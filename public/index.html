const EP_LIST = [ { id: 'gemini', category: 'ai', title: 'Gemini AI', method: 'GET', path: '/api/gemini', result: { type: 'text', field: 'text' }, params: [ { name: 'q', type: 'textarea', label: 'Pertanyaan', required: true, placeholder: 'Tulis pertanyaan…', value: '' } ] }, { id: 'pinterest', category: 'search', title: 'Pinterest Search', method: 'GET', path: '/api/search/pinterest', result: { type: 'text', field: 'count' }, params: [ { name: 'q', type: 'textarea', label: 'Kata Kunci', required: true, placeholder: 'Contoh: anime, kucing', value: '' } ] }, { id: 'tiktok', category: 'download', title: 'TikTok Search', method: 'GET', path: '/api/download/tiktok', result: { type: 'text', field: 'count' }, params: [ { name: 'q', type: 'textarea', label: 'Kata Kunci', required: true, placeholder: 'Contoh: kucing lucu', value: '' } ] } ]; const LABELS = { ai: 'AI', search: 'Pencarian', download: 'Unduhan' }; const el = (tag, cls, text) => { const n = document.createElement(tag); if (cls) n.className = cls; if (text != null) n.textContent = text; return n; }; const cap = (s) => s.charAt(0).toUpperCase() + s.slice(1); const sh = (s) => "'" + s.replace(/'/g, "'\\''") + "'"; function render() { const app = document.getElementById('app'); app.innerHTML = ''; const nav = el('nav'); const h1 = el('h1', null, 'BmskyAPIs'); nav.append(h1); app.append(nav); const groups = {}; EP_LIST.forEach(ep => { if (!groups[ep.category]) groups[ep.category] = []; groups[ep.category].push(ep); }); for (const [cat, eps] of Object.entries(groups)) { const sec = el('section', 'cat'); sec.append(el('h2', 'cat-title', LABELS[cat] || cap(cat))); eps.forEach(ep => { const wrap = el('div', 'ep-wrap'); const btn = el('button', 'ep-btn', ep.title); wrap.append(btn); sec.append(wrap); let pg = null; btn.addEventListener('click', () => { document.querySelectorAll('.ep-wrap.open').forEach(w => { if (w !== wrap) w.classList.remove('open'); }); wrap.classList.toggle('open'); if (!pg) { pg = createPlayground(ep); wrap.append(pg); } }); }); app.append(sec); } } function createPlayground(ep) { const values = {}; let busy = false, cooldown = null; const root = el('div', 'pg'); root.innerHTML = '
' + '' + '

' + '
' + '
Buka hasil
' + '

' + '

' +
        '

Belum ada hasil.

      '
' +
    '
';

  const q = (sel) => root.querySelector(sel);
  const runBtn = q('.run'), hintEl = q('.hint');
  const chip = q('.chip'), dl = q('.dl'), out = q('.out'), curlEl = q('.curl');

  const val = (name) => String(values[name] || '').replace(/\s+/g, ' ').trim();
  const hint = (t) => { hintEl.textContent = t || ''; };
  const setState = (s) => { root.dataset.state = s; };

  ep.params.forEach(p => {
    const wrap = el('div', 'f');
    const id = ep.id + '-' + p.name;
    const lab = el('label', null, p.label + (p.required ? '' : ' (opsional)'));
    lab.htmlFor = id;
    let inp;
    if (p.type === 'select') {
      inp = el('select');
      (p.options || []).forEach(opt => {
        const o = el('option', null, opt.label || opt);
        o.value = opt.value || opt;
        if (opt.value === p.value) o.selected = true;
        inp.append(o);
      });
    } else if (p.type === 'textarea') {
      inp = el('textarea');
      inp.rows = 3;
    } else {
      inp = el('input');
      inp.type = p.type || 'text';
    }
    inp.id = id;
    inp.placeholder = p.placeholder || '';
    inp.value = p.value || '';
    values[p.name] = inp.value;

    let counter = null;
    if (p.max) {
      counter = el('span', 'count', inp.value.length + '/' + p.max);
    }
    inp.addEventListener('input', () => {
      values[p.name] = inp.value;
      if (counter) counter.textContent = inp.value.length + '/' + p.max;
      updateCurl();
    });
    if (p.type === 'textarea') {
      inp.addEventListener('keydown', ev => {
        if (ev.key === 'Enter' && (ev.ctrlKey || ev.metaKey)) { ev.preventDefault(); run(); }
      });
    }
    wrap.append(lab, inp);
    if (counter) wrap.append(counter);
    q('.fields').append(wrap);
  });

  function queryString() {
    const s = new URLSearchParams();
    ep.params.forEach(p => { const v = val(p.name); if (v) s.set(p.name, v); });
    return s.toString();
  }

  function updateCurl() {
    const params = ep.params.filter(p => val(p.name)).map(p => p.name + '=' + encodeURIComponent(val(p.name))).join('&');
    curlEl.textContent = location.origin + ep.path + (params ? '?' + params : '');
  }

  function setButton(label, disabled) { runBtn.textContent = label; runBtn.disabled = disabled; }
  function startCooldown(sec) {
    clearInterval(cooldown);
    let left = Math.max(1, Math.ceil(sec));
    setButton('Tunggu ' + left + ' dtk', true);
    cooldown = setInterval(() => {
      left--;
      if (left <= 0) { clearInterval(cooldown); setButton('Kirim permintaan', false); return; }
      setButton('Tunggu ' + left + ' dtk', true);
    }, 1000);
  }

  async function run() {
    if (busy) return;
    busy = true;
    setState('loading');
    setButton('Memproses…', true);
    chip.textContent = '';
    chip.className = 'chip';
    dl.hidden = true;
    out.textContent = 'Memproses permintaan…';
    hint('');

    const t0 = performance.now();
    let status = 0, json = null, retry = 0;
    try {
      const r = await fetch(ep.path + '?' + queryString(), { headers: { accept: 'application/json' } });
      status = r.status;
      retry = Number(r.headers.get('retry-after')) || 0;
      const txt = await r.text();
      try { json = JSON.parse(txt); }
      catch (e) { json = { success: false, error: { message: 'Respons tidak valid: ' + txt.slice(0, 200) } }; }
    } catch (e) {
      json = { success: false, error: { message: 'Tidak terhubung ke server.' } };
    }
    const took = ((performance.now() - t0) / 1000).toFixed(1);

    // Tampilkan JSON lengkap
    out.textContent = JSON.stringify(json, null, 2);

    if (json && json.success) {
      setState('done');
      chip.className = 'chip ok';
      chip.textContent = status + ' · ' + took + ' dtk';
      // Coba cari link unduhan
      const data = json.data;
      if (data) {
        const findUrl = (obj) => {
          if (!obj) return null;
          if (typeof obj === 'string' && obj.startsWith('http')) return obj;
          if (Array.isArray(obj)) return findUrl(obj[0]);
          if (typeof obj === 'object') {
            for (const v of Object.values(obj)) {
              const u = findUrl(v);
              if (u) return u;
            }
          }
          return null;
        };
        const url = findUrl(data);
        if (url) { dl.href = url; dl.hidden = false; }
      }
      startCooldown(5);
    } else {
      const m = (json && json.error && json.error.message) || 'Permintaan gagal.';
      setState('error');
      chip.className = 'chip bad';
      chip.textContent = (status || 'ERR') + ' · ' + took + ' dtk';
      hint(m);
      startCooldown(status === 429 ? (retry || 60) : 3);
    }
    busy = false;
  }

  runBtn.addEventListener('click', run);
  q('.copy').addEventListener('click', async ev => {
    const b = ev.currentTarget;
    try { await navigator.clipboard.writeText(curlEl.textContent); b.textContent = 'Tersalin'; }
    catch (e) { b.textContent = 'Gagal menyalin'; }
    setTimeout(() => { b.textContent = 'Salin'; }, 1500);
  });

  updateCurl();
  return root;
}
